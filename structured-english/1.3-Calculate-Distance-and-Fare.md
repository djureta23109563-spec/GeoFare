# 1.3 Calculate Distance & Fare

```text
BEGIN

    GET Pickup_Location
    GET Dropoff_Location

    CALCULATE Trip_Distance using GPS_Coordinates

    GET Official_Base_Fare from Fare_Rate_DB
    GET Per_Kilometer_Rate from Fare_Rate_DB

    IF Trip_Distance is within the Base_Fare coverage THEN
        SET Estimated_Fare = Official_Base_Fare
    ELSE
        CALCULATE Additional_Distance
        CALCULATE Additional_Fare using Per_Kilometer_Rate
        SET Estimated_Fare = Official_Base_Fare + Additional_Fare
    ENDIF

    DISPLAY Estimated_Fare to Passenger

    STORE Trip_Distance and Estimated_Fare in Trip_Logs

END
